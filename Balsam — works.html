# Clean the survey and compare priorities across faculties
import pandas as pd
from scipy.stats import chi2_contingency

df = pd.read_csv("student_survey.csv")
df = df.drop_duplicates("response_id").dropna(subset=["priority"])
df["faculty"] = df["faculty"].str.strip().str.title()

table = pd.crosstab(df["faculty"], df["priority"])
chi2, p, dof, exp = chi2_contingency(table)
print(table.div(table.sum(axis=1), axis=0).round(2), p)
